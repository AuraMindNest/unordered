=== `<boost/unordered/unordered_set.hpp>` 概要

定义 link:reference/unordered_set_zh_Hans.adoc#unordered_set[`boost::unordered_set`]、link:reference/unordered_multiset_zh_Hans.adoc#unordered_multiset[`boost::unordered_multiset`] 以及相关的函数和别名模板。

namespace boost {
namespace unordered {

  template<class Key,
           class Hash = boost::hash<Key>,
           class Pred = std::equal_to<Key>,
           class Allocator = std::allocator<Key>>
  class link:reference/unordered_set_zh_Hans.adoc#unordered_set[unordered_set];

  // 相等比较
  template<class Key, class Hash, class Pred, class Alloc>
    bool link:reference/unordered_set_zh_Hans.adoc#unordered_set_operator[operator==](const unordered_set<Key, Hash, Pred, Alloc>&amp; x,
                    const unordered_set<Key, Hash, Pred, Alloc>&amp; y);

  template<class Key, class Hash, class Pred, class Alloc>
    bool link:reference/unordered_set_zh_Hans.adoc#unordered_set_operator_2[operator!=](const unordered_set<Key, Hash, Pred, Alloc>&amp; x,
                    const unordered_set<Key, Hash, Pred, Alloc>&amp; y);

  // 交换
  template<class Key, class Hash, class Pred, class Alloc>
    void link:reference/unordered_set_zh_Hans.adoc#unordered_set_swap_2[swap](unordered_set<Key, Hash, Pred, Alloc>&amp; x,
              unordered_set<Key, Hash, Pred, Alloc>&amp; y)
      noexcept(noexcept(x.swap(y)));

  // 擦除
  template<class K, class H, class P, class A, class Predicate>
    typename unordered_set<K, H, P, A>::size_type
      link:reference/unordered_set_zh_Hans.adoc#unordered_set_erase_if[erase_if](unordered_set<K, H, P, A>&amp; c, Predicate pred);

  template<class Key,
           class Hash = boost::hash<Key>,
           class Pred = std::equal_to<Key>,
           class Allocator = std::allocator<Key>>
  class link:reference/unordered_multiset_zh_Hans.adoc#unordered_multiset[unordered_multiset];

  // 相等比较
  template<class Key, class Hash, class Pred, class Alloc>
    bool link:reference/unordered_multiset_zh_Hans.adoc#unordered_multiset_operator[operator==](const unordered_multiset<Key, Hash, Pred, Alloc>&amp; x,
                    const unordered_multiset<Key, Hash, Pred, Alloc>&amp; y);

  template<class Key, class Hash, class Pred, class Alloc>
    bool link:reference/unordered_multiset_zh_Hans.adoc#unordered_multiset_operator_2[operator!=](const unordered_multiset<Key, Hash, Pred, Alloc>&amp; x,
                    const unordered_multiset<Key, Hash, Pred, Alloc>&amp; y);

  // 交换
  template<class Key, class Hash, class Pred, class Alloc>
    void link:reference/unordered_multiset_zh_Hans.adoc#unordered_multiset_swap_2[swap](unordered_multiset<Key, Hash, Pred, Alloc>&amp; x,
              unordered_multiset<Key, Hash, Pred, Alloc>&amp; y)
      noexcept(noexcept(x.swap(y)));

  // 擦除
  template<class K, class H, class P, class A, class Predicate>
    typename unordered_multiset<K, H, P, A>::size_type
      link:reference/unordered_multiset_zh_Hans.adoc#unordered_multiset_erase_if[erase_if](unordered_multiset<K, H, P, A>&amp; c, Predicate pred);

  // Pmr别名（C++17及以上）
  namespace pmr {
    template<class Key,
             class Hash = boost::hash<Key>,
             class Pred = std::equal_to<Key>>
    using unordered_set =
      boost::unordered::unordered_set<Key, Hash, Pred,
        std::pmr::polymorphic_allocator<Key>>;

    template<class Key,
             class Hash = boost::hash<Key>,
             class Pred = std::equal_to<Key>>
    using unordered_multiset =
      boost::unordered::unordered_multiset<Key, Hash, Pred,
        std::pmr::polymorphic_allocator<Key>>;
  } // 命名空间 pmr

} // 命名空间 unordered

using unordered::unordered_set;
using unordered::unordered_multiset;

} // 命名空间 boost
